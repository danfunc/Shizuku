cmake_minimum_required(VERSION 3.26)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fPIE -fPIC")
set(CMAKE_CXX_STANDARD "20")
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/buildUtilities.cmake)
preloadToolchain()
project(Shizuku C CXX ASM)
add_executable(Shizuku)
add_library(Shizuku_HEADERS INTERFACE)
target_include_directories(Shizuku_HEADERS INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/interface")
target_link_libraries(Shizuku PUBLIC Shizuku_HEADERS)
postloadToolchain()
add_subdirectory(modules)
add_subdirectory(configs)
add_subdirectory(source)
